### 简易版响应式 Vue2

```JS
/**
 * 观察某个对象的属性变化，简易版响应式数据
 * @param {Object} obj
 */
function observe(obj) {
  for (const key in obj) {
    let innerValue = obj[key];
    let funcs = new Set();
    Object.defineProperty(obj, key, {
      get() {
        // 收集依赖: 将当前函数放入依赖集合中
        if (window.__fn) {
          funcs.add(window.__fn);
        }
        return innerValue;
      },
      set(newVal) {
        // 触发依赖: 执行依赖集合中的所有函数
        innerValue = newVal;
        for (const fn of funcs) {
          fn();
        }
      },
    });
  }
}

/**
 * 自动运行函数
 * @param {function} fn
 */
function autorun(fn) {
  window.__fn = fn;
  fn();
  window.__fn = null;
}

```

### vue3 响应式简易版

```js
let data = {
        A1: 0,
        A2: 0,
      };
      function update() {
        let sum = data.A1 + data.A2;
        console.log(sum);
      }

      function wentDepsChange(callback) {
        const proxyData = new Proxy(data, {
        //监听对象的变化 从而触发回调函数 从而实现响应式
          //target是目标对象 key是属性名 value是属性值 receiver是代理对象
          set(target, key, value, receiver) {
            target[key] = value;
            callback(); //更新
          },
          get(target, key, receiver) {
            return target[key];
          },
        });
        return proxyData;
      }
      const proxyData = wentDepsChange(update);
      setInterval(() => {
        proxyData.A1++;
      }, 500);
```
