# 函数防抖

```js
      function debounce(fn, duration = 500) {
        var timeId = null;
        return function () {
          clearTimeout(timeId);
          //将函数的this保存
          var that = this;
          //Array.prototype.slice this指向Array.prototype
          //将函数的参数获取，将slice的指向修改成arguments
          var args = Array.prototype.slice.call(arguments, 0);
          timeId = setTimeout(function () {
            fn.apply(that, args);
          }, duration);
        };
      }
    //使用
	const handle = debounce(function (e) {
        console.log(e, "--", this.value);
    });
    inp.addEventListener("keyup", handle);
```
