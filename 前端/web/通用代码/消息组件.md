# 消息组件

```js
// showMessage.js
import Icon from "@/components/Icon.vue";
import Vue from "vue";
import styles from "./showMessage.module.scss";

//是否正在显示消息
let messageInstance = false;

/**
 * 获取某个组件渲染的Dom根元素
 * @param {Component} comp 组件
 * @param {Object} props 配置
 * @returns dom
 */
function getComponentRootDom(comp, props) {
  const vm = new Vue({
    render: (h) => h(comp, { props }),
  });
  vm.$mount();
  return vm.$el;
}

/**
 * 消息提示
 * @param {Object} options 配置项
 * @param {String} options.content 消息内容
 * @param {String} options.type error/success/warning/info 消息类型
 * @param {number} options.duration 消失时间，0不会消失
 * @param {HTMLElement} options.container 容器，消息会显示在容器中间，如果不传则显示在页面中间
 * @param {Function} options.callback 回调函数
 */
export default function (options) {
  let { content, type, duration, container } = options;
  type = type || "info";
  duration = duration || 2000;
  container = container || document.body;
  if (messageInstance) return;
  messageInstance = true;
  // 创建一个div元素
  const div = document.createElement("div");
  // 获取icon组件渲染的Dom根元素
  const iconDom = getComponentRootDom(Icon, { type });
  div.innerHTML = `<span class=${styles.icon}>${iconDom.outerHTML}</span><div>${content}</div>`;

  div.className = `${styles.message} ${styles["message-" + type]}`;

  if (getComputedStyle(container).position === "static") {
    container.style.position = "relative";
  }
  container.appendChild(div);

  // 浏览器强行渲染
  div.clientHeight; // 重绘

  // 进入动画
  div.style.opacity = 1;
  div.style.transform = `translate(-50%, -50%)`;

  setTimeout(() => {
    div.style.opacity = 0;
    div.style.transform = `translate(-50%, -50%) translateY(-10px)`;
    // 监听动画结束事件 移除dom 执行一次
    div.addEventListener(
      "transitionend",
      function () {
        div.remove();
        messageInstance = false;
        // 执行回调
        if (options.callback) options.callback && options.callback();
      },
      { once: true }
    );
  }, duration);
}
```

```vue
<!-- Icon.vue  -->
<template>
  <i class="iconfont" :class="fontClass"></i>
</template>
<script>
const iconMap = {
  error: "icon-error",
  success: "icon-success",
  warning: "icon-warning",
  info: "icon-info",
};
export default {
  props: {
    type: {
      type: String,
      required: true,
    },
  },
  computed: {
    fontClass() {
      return iconMap[this.type];
    },
  },
};
</script>
<style lang="scss" scoped>
@import "@/assets/iconfont/iconfont.css";
.iconfont {
  font-size: inherit;
  color: inherit;
}
</style>
```
