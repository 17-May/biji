### 初体验

```jsx
  return (
    <div className="App">
      <ul className="url_wrap">
        <li className="item">商城</li>
        <li className="item">购物车</li>
        <li className="item">订单</li>
        <li className="item">
          <span>订单</span>
        </li>
      </ul>
    </div>
  );
```

```js
return React.createElement(
    "div",
    { className: "App" },
    React.createElement(
      "ul",
      { className: "url_wrap" },
      React.createElement("li", { className: "item" }, "商城"),
      React.createElement("li", { className: "item" }, "购物车"),
      React.createElement("li", { className: "item" }, "订单"),
      React.createElement(
        "li",
        { className: "item" },
        React.createElement("span", null, "我的")
      )
    )
  );
```
