# 点击随机方块

# 步骤

- 顶点着色器

  - 设置顶点着色器xy轴变量和宽高变量
  - 设置计算顶点着色器canvas与webgl转换公式，并赋值
- 片源着色器

  - 设置片源着色器颜色变量
  - 设置颜色转换，并赋值
- Js

  - 获取xy轴、canvas宽高和随机的颜色
  - 获取顶点着色器和片源着色器变量地址，并为顶点着色器宽高变量赋值
  - 初始化画布
  - 使用数组保存生成的方块
  - 每次点击都需要初始化画布，循环数组并分别对变量赋值

### index.vert

```.vert
// 定义浮点精度
precision mediump float;

// 定义变量
// 维度 变量名
// x y轴坐标
attribute vec2 a_position;
// 屏幕宽高
attribute vec2 a_screen_size;
// canvas和webgl 转换公式
// (x/canvas宽度) * 2.0 - 1.0
// (y/canvas高度) * 2.0 - 1.0
void main() {
  // (300,150)/(300,150) = (1,1)
  vec2 position = a_position / a_screen_size * 2.0 - 1.0;
  // 翻转y轴 canvas y轴向下 webgl y轴向上
  position = position * vec2(1.0, -1.0);
  // xyz坐标轴 0为坐标原点
  // -1.0~1.0
  gl_Position = vec4(position, 0.0, 1.0);
  // 顶点大小
  gl_PointSize = 10.0;
}
```

### index.frag

```.frag
precision mediump float;
// rgba颜色
uniform vec4 u_FragColor;

void main() {
  vec4 color = u_FragColor / vec4(255, 255, 255, 1);
  gl_FragColor = color;
}
```

### index.js

```JavaScript
import frag from "./glsl/index.frag";
import vert from "./glsl/index.vert";
import index from "./util/index";
const { cearteShader, cearteProgram, randomColor } = index;
/** @type{HTMLCanvasElement} */
const canvas = document.querySelector("#canvas");
const gl =
  canvas.getContext("webgl") || canvas.getContext("experimental-webgl");

// 流程1 创建顶点着色器
const vertexShader = cearteShader(gl, gl.VERTEX_SHADER, vert);
// 流程2 创建片元着色器
const fragmentShader = cearteShader(gl, gl.FRAGMENT_SHADER, frag);
// 流程3 创建着色器程序
const program = cearteProgram(gl, vertexShader, fragmentShader);
// 链接成功后使用
gl.useProgram(program);

// 获取变量地址
const a_position = gl.getAttribLocation(program, "a_position");
const a_screen_size = gl.getAttribLocation(program, "a_screen_size");
const u_FragColor = gl.getUniformLocation(program, "u_FragColor");

// 赋值
gl.vertexAttrib2f(a_screen_size, canvas.width, canvas.height);
let points = [];
canvas.addEventListener("click", (e) => {
  const x = e.clientX;
  const y = e.clientY;
  const color = randomColor();
  points.push({
    x,
    y,
    color,
  });
  initGL();
  points.forEach((item) => {
    const { x, y, color } = item;
    gl.uniform4f(u_FragColor, color.r, color.g, color.b, color.a);
    gl.vertexAttrib2f(a_position, x, y);
    gl.drawArrays(gl.POINTS, 0, 1);
  });
});
function initGL() {
  // 清除画布 并设置初始颜色
  gl.clearColor(0.0, 0.0, 0.0, 1.0);
  // 用上一步设置的颜色填充画布
  gl.clear(gl.COLOR_BUFFER_BIT);
}
initGL();
```

### utils/index.js

```JavaScript
/**
 * 创建着色器
 * @param {*} gl gl对象
 * @param {*} type 着色器类型
 * @param {*} source 内容
 * @returns 着色器对象
 */
function cearteShader(gl, type, source) {
  const shader = gl.createShader(type);
  gl.shaderSource(shader, source);
  gl.compileShader(shader);
  return shader;
}

/**
 * 创建着色器程序
 * @param {*} gl gl对象
 * @param {*} vertexShader 顶点着色器
 * @param {*} fragmentShader 片元着色器
 * @returns 着色器程序
 */
function cearteProgram(gl, vertexShader, fragmentShader) {
  const program = gl.createProgram();
  gl.attachShader(program, vertexShader);
  gl.attachShader(program, fragmentShader);
  gl.linkProgram(program);
  return program;
}
/**生成随机rgba颜色 */
function randomColor() {
  return {
    r: Math.random() * 255,
    g: Math.random() * 255,
    b: Math.random() * 255,
    a: Math.random(),
  };
}
export default { cearteShader, cearteProgram, randomColor };
```
