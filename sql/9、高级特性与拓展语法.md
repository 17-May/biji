### 9.1 预处理语句 PREPARE

``` sql
-- 1. 定义预处理  sql 模板
SET @ sql_template = 'SELECT * FROM 表名 WHERE id = ?';

-- 2. 预处理编译语句
PREPARE stmt FROM @ sql_template;

-- 3. 绑定参数并执行
SET @id = 1;
EXECUTE stmt USING @id;

-- 4. 释放预处理语句
DEALLOCATE PREPARE stmt;
```

### 9.2 表复制、数据导入导出

#### 9.2.1 表复制

``` sql
-- 仅复制表结构，不复制数据
CREATE TABLE 新表名 LIKE 原表名;

-- 复制表结构 + 全部数据
CREATE TABLE 新表名 SELECT * FROM 原表名;

-- 复制指定字段与筛选后数据
CREATE TABLE 新表名 SELECT 字段1,字段2 FROM 原表名 WHERE 条件;
```

#### 9.2.2 数据导入

``` sql
-- 文本文件导入表数据
LOAD DATA [LOCAL] INFILE '文件路径'
INTO TABLE 表名
[CHARACTER SET 字符集]
FIELDS
TERMINATED BY '字段分隔符'
[ENCLOSED BY '包裹符']
LINES
STARTING BY '行起始符'
TERMINATED BY '行结束符';
```

#### 9.2.3 数据导出

``` sql
-- 查询结果导出为文本文件
SELECT 字段列表 FROM 表名
INTO OUTFILE '保存路径'
[CHARACTER SET 字符集]
FIELDS
TERMINATED BY '字段分隔符'
LINES
TERMINATED BY '行结束符';
```

### 9.3 元数据查询（information_schema 系统库）

``` sql
-- 查询当前实例所有数据库
SELECT SCHEMA_NAME FROM information_schema.SCHEMATA;

-- 查询指定库下所有数据表
SELECT TABLE_NAME FROM information_schema.TABLES 
WHERE TABLE_SCHEMA = '数据库名';

-- 查询表字段信息
SELECT COLUMN_NAME,DATA_TYPE,COLUMN_COMMENT 
FROM information_schema.COLUMNS 
WHERE TABLE_SCHEMA = '数据库名' AND TABLE_NAME = '表名';

-- 查询索引信息
SELECT INDEX_NAME,COLUMN_NAME 
FROM information_schema.STATISTICS 
WHERE TABLE_SCHEMA = '数据库名' AND TABLE_NAME = '表名';

-- 查询视图
SELECT TABLE_NAME FROM information_schema.VIEWS 
WHERE TABLE_SCHEMA = '数据库名';

-- 查询存储过程/函数
SELECT ROUTINE_NAME,ROUTINE_TYPE 
FROM information_schema.ROUTINES 
WHERE ROUTINE_SCHEMA = '数据库名';
```

### 9.4 全文检索、JSON 数据操作

#### 9.4.1 全文检索

``` sql
-- 基于全文索引检索
SELECT * FROM 表名 
WHERE MATCH(全文索引字段) AGAINST('检索关键词');

-- 布尔模式检索
SELECT * FROM 表名 
WHERE MATCH(字段) AGAINST('+关键词1 -关键词2' IN BOOLEAN MODE);
```

#### 9.4.2 JSON 数据操作

``` sql
-- 1. 插入 JSON 数据
INSERT INTO 表名(json字段) VALUES('{"name":"张三","age":20}');

-- 2. 读取 JSON 字段值
SELECT json字段->'$.name' FROM 表名;
SELECT json字段->>'$.name' FROM 表名; -- 去除引号

-- 3. 修改 JSON 数据
UPDATE 表名 
SET json字段 = JSON_SET(json字段, '$.age', 21) 
WHERE 条件;

-- 4. 追加 JSON 键值对
UPDATE 表名 
SET json字段 = JSON_INSERT(json字段, '$.sex', '男') 
WHERE 条件;

-- 5. 删除 JSON 指定节点
UPDATE 表名 
SET json字段 = JSON_REMOVE(json字段, '$.sex') 
WHERE 条件;

-- 6. 判断 JSON 是否包含指定数据
SELECT * FROM 表名 
WHERE JSON_CONTAINS(json字段, '{"age":20}');
```

### 9.5 公用表表达式 CTE（My sql 8.0+ 支持）

#### 9.5.1 普通 CTE

``` sql
-- 单 CTE
WITH cte_name AS (
    SELECT 字段 FROM 表名 WHERE 条件
)
SELECT * FROM cte_name;

-- 多 CTE（逗号分隔）
WITH 
cte1 AS (SELECT 字段 FROM 表1),
cte2 AS (SELECT 字段 FROM 表2)
SELECT * FROM cte1 JOIN cte2 ON 关联条件;
```

#### 9.5.2 递归 CTE

``` sql
-- 语法框架（常用于树形结构、层级查询）
WITH RECURSIVE cte_name(字段列表) AS (
    -- 锚点成员：初始数据
    SELECT 字段 FROM 表名 WHERE 根节点条件
    UNION ALL
    -- 递归成员：关联自身迭代查询
    SELECT 字段 FROM 表名 t
    INNER JOIN cte_name c ON t.关联字段 = c.关联字段
)
SELECT * FROM cte_name;
```
