### 7.1 系统内置函数

#### 7.1.1 字符串函数

``` sql
-- 拼接字符串
CONCAT(str1,str2,...)
-- 指定分隔符拼接
CONCAT_WS(分隔符,str1,str2,...)
-- 获取字符串长度
LENGTH(str)
CHAR_LENGTH(str)
-- 大小写转换
UPPER(str) / UCASE(str)
LOWER(str) / LCASE(str)
-- 截取字符串
SUBSTR(str,pos)
SUBSTRING(str,pos,len)
-- 去除首尾空格
TRIM(str)
LTRIM(str)
RTRIM(str)
-- 替换字符串
REPLACE(str,old_str,new_str)
-- 字符串查找
INSTR(str,substr)
-- 重复字符串
REPEAT(str,count)
-- 反转字符串
REVERSE(str)
```

#### 7.1.2 数值函数

``` sql
-- 取绝对值
ABS(num)
-- 向上取整
CEIL(num) / CEILING(num)
-- 向下取整
FLOOR(num)
-- 四舍五入
ROUND(num,保留位数)
-- 取模运算
MOD(num1,num2)
-- 随机数 [0,1)
RAND()
-- 幂运算
POW(num,次方) / POWER(num,次方)
-- 平方根
SQRT(num)
-- 取最大值、最小值
GREATEST(val1,val2,...)
LEAST(val1,val2,...)
```

#### 7.1.3 日期时间函数

``` sql
-- 获取当前日期+时间
NOW() / SYSDATE() / CURRENT_TIMESTAMP
-- 获取当前日期
CURDATE() / CURRENT_DATE
-- 获取当前时间
CURTIME() / CURRENT_TIME
-- 提取年月日时分秒
YEAR(date)
MONTH(date)
DAY(date)
HOUR(time)
MINUTE(time)
SECOND(time)
-- 日期加减
DATE_ADD(date,INTERVAL 数值 单位)
DATE_SUB(date,INTERVAL 数值 单位)
-- 日期差值
DATEDIFF(date1,date2)
TIMESTAMPDIFF(单位,时间1,时间2)
-- 日期格式化
DATE_FORMAT(date,格式符)
TIME_FORMAT(time,格式符)
```

#### 7.1.4 流程控制函数

``` sql
-- 条件判断
IF(条件,成立值,不成立值)
-- 空值判断
IFNULL(表达式,默认值)
NULLIF(表达式1,表达式2)
-- 多分支判断
CASE 
WHEN 条件1 THEN 结果1
WHEN 条件2 THEN 结果2
ELSE 其他结果
END
-- 等值匹配CASE
CASE 字段
WHEN 值1 THEN 结果1
WHEN 值2 THEN 结果2
ELSE 其他结果
END
```

#### 7.1.5 聚合函数

``` sql

-- 求和
SUM(字段)
-- 平均值
AVG(字段)
-- 最大值
MAX(字段)
-- 最小值
MIN(字段)
```

#### 7.1.6 加密、系统信息函数


### 7.2 自定义函数（UDF）

#### 7.2.1 创建自定义函数

``` sql
-- 语法结构
DELIMITER 新结束符
CREATE FUNCTION 函数名(参数名 参数类型) 
RETURNS 返回值类型
[函数特性]
BEGIN
    -- 函数逻辑体
    RETURN 返回数据;
END
新结束符
DELIMITER ; -- 恢复默认结束符

-- 示例：两数求和
DELIMITER $$
CREATE FUNCTION add_num(a INT, b INT)
RETURNS INT
DETERMINISTIC
BEGIN
    RETURN a + b;
END $$
DELIMITER ;
```

#### 7.2.2 调用自定义函数

``` sql
-- 单行调用
SELECT 函数名(实参);
-- 结合表查询调用
SELECT 字段,函数名(字段) FROM 表名;

-- 示例调用
SELECT add_num(10,20);
```

#### 7.2.3 查看自定义函数

``` sql
-- 查看指定数据库下所有函数
SHOW FUNCTION STATUS WHERE Db = '数据库名';
-- 查看函数创建语句
SHOW CREATE FUNCTION 函数名;
```

#### 7.2.4 删除自定义函数

``` sql
-- 直接删除
DROP FUNCTION 函数名;
-- 存在则删除，避免报错
DROP FUNCTION IF EXISTS 函数名;
```
